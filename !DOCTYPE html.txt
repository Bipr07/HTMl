<script>
/* =========================================================
   EmailJS SETUP — replace these three values with your own
   from https://www.emailjs.com (free plan works fine).
   1. Create an Email Service  -> copy the Service ID
   2. Create an Email Template -> copy the Template ID

  const EMAILJS_SERVICE_ID  = "YOUR_SERVICE_ID";
  const EMAILJS_TEMPLATE_ID = "YOUR_TEMPLATE_ID";
  const EMAILJS_PUBLIC_KEY  = "YOUR_PUBLIC_KEY";
  const SHOP_OWNER_EMAIL    = "joym9047@gmail.com";

  emailjs.init(EMAILJS_PUBLIC_KEY);

  // ---- Product data ----
  const PRODUCTS = [
    { id: 1, name: "Farmhouse Basket", price: 349, img: "https://i.ibb.co/Rk0kbNJW/0468c649ab6656523dfe762ed5e7330a.jpg" },
    { id: 2, name: "Daily Essentials Pack", price: 279, img: "https://i.ibb.co/TMBv9HVW/15346be7e71d7366824e3c9b59651ef0.jpg" },
    { id: 3, name: "Pantry Staples Box", price: 459, img: "https://i.ibb.co/b5tyZT0B/70748ffd66fdacd9fd39f068a323d68f.jpg" },
    { id: 4, name: "Fresh Picks Bundle", price: 399, img: "https://i.ibb.co/fGP2mdtW/40b9ba00b5928c73594aa0fcf41033c7.jpg" },
  ];

  let cart = {}; // { id: qty }

  // ---- Render product carousel ----
  const carouselEl = document.getElementById('carousel');
  PRODUCTS.forEach(p => {
    const card = document.createElement('div');
    card.className = 'card';
    card.innerHTML = `
      <img class="card-img" src="${p.img}" alt="${p.name}" loading="lazy" />
      <div class="card-body">
        <h3>${p.name}</h3>
        <div class="card-price">₹${p.price}</div>
        <button class="card-add" data-id="${p.id}">Add to Cart</button>
      </div>`;
    carouselEl.appendChild(card);
  });

  carouselEl.addEventListener('click', (e) => {
    const btn = e.target.closest('.card-add');
    if (!btn) return;
    const id = Number(btn.dataset.id);
    cart[id] = (cart[id] || 0) + 1;
    renderCart();
    btn.textContent = "Added ✓";
    btn.classList.add('added');
    setTimeout(() => { btn.textContent = "Add to Cart"; btn.classList.remove('added'); }, 900);
  });

  // ---- Cart drawer logic ----
  const drawer = document.getElementById('drawer');
  const overlay = document.getElementById('overlay');
  const cartBtn = document.getElementById('cartBtn');
  const drawerClose = document.getElementById('drawerClose');
  const drawerItems = document.getElementById('drawerItems');
  const cartCountEl = document.getElementById('cartCount');
  const subtotalEl = document.getElementById('subtotal');
  const formMsg = document.getElementById('formMsg');

  function openDrawer(){ drawer.classList.add('open'); overlay.classList.add('open'); }
  function closeDrawer(){ drawer.classList.remove('open'); overlay.classList.remove('open'); }
  cartBtn.addEventListener('click', openDrawer);
  drawerClose.addEventListener('click', closeDrawer);
  overlay.addEventListener('click', closeDrawer);

  function renderCart(){
    const ids = Object.keys(cart).filter(id => cart[id] > 0);
    const totalCount = ids.reduce((sum, id) => sum + cart[id], 0);
    cartCountEl.textContent = totalCount;

    if (ids.length === 0){
      drawerItems.innerHTML = `<div class="drawer-empty">Your basket is empty.<br>Add something fresh from the shelf.</div>`;
      subtotalEl.textContent = "₹0";
      return;
    }

    let total = 0;
    drawerItems.innerHTML = ids.map(id => {
      const p = PRODUCTS.find(pr => pr.id === Number(id));
      const qty = cart[id];
      const lineTotal = p.price * qty;
      total += lineTotal;
      return `
        <div class="drawer-item" data-id="${p.id}">
          <img src="${p.img}" alt="${p.name}" />
          <div class="drawer-item-info">
            <h4>${p.name}</h4>
            <div class="item-price">₹${lineTotal}</div>
            <div class="qty-controls">
              <button class="qty-minus" data-id="${p.id}">−</button>
              <span>${qty}</span>
              <button class="qty-plus" data-id="${p.id}">+</button>
            </div>
            <button class="remove-btn" data-id="${p.id}">Remove</button>
          </div>
        </div>`;
    }).join('');
    subtotalEl.textContent = "₹" + total;
  }

  drawerItems.addEventListener('click', (e) => {
    const id = e.target.dataset.id;
    if (!id) return;
    if (e.target.classList.contains('qty-plus')) cart[id]++;
    if (e.target.classList.contains('qty-minus')) cart[id] = Math.max(0, cart[id] - 1);
    if (e.target.classList.contains('remove-btn')) delete cart[id];
    renderCart();
  });

  // ---- Order form / EmailJS send ----
  const orderForm = document.getElementById('orderForm');
  const sendBtn = document.getElementById('sendBtn');

  orderForm.addEventListener('submit', function(e){
    e.preventDefault();
    formMsg.textContent = "";
    formMsg.className = "form-msg";

    const ids = Object.keys(cart).filter(id => cart[id] > 0);
    if (ids.length === 0){
      formMsg.textContent = "Add at least one item before sending your order.";
      formMsg.className = "form-msg error";
      return;
    }

    const name = document.getElementById('custName').value.trim();
    const phone = document.getElementById('custPhone').value.trim();

    let total = 0;
    const itemLines = ids.map(id => {
      const p = PRODUCTS.find(pr => pr.id === Number(id));
      const qty = cart[id];
      total += p.price * qty;
      return `${p.name} x${qty} — ₹${p.price * qty}`;
    });

    const templateParams = {
      customer_name: name,
      customer_phone: phone,
      order_items: itemLines.join("\n"),
      order_total: "₹" + total,
      to_email: SHOP_OWNER_EMAIL
    };

    sendBtn.disabled = true;
    sendBtn.textContent = "Sending...";

    emailjs.send(EMAILJS_SERVICE_ID, EMAILJS_TEMPLATE_ID, templateParams)
      .then(() => {
        formMsg.textContent = "Order sent! We'll confirm with you shortly.";
        formMsg.className = "form-msg success";
        cart = {};
        renderCart();
        orderForm.reset();
      })
      .catch((err) => {
        console.error(err);
        formMsg.textContent = "Couldn't send the order. Please try again or WhatsApp us directly.";
        formMsg.className = "form-msg error";
      })
      .finally(() => {
        sendBtn.disabled = false;
        sendBtn.textContent = "Send Order";
      });
  });

  document.getElementById('year').textContent = new Date().getFullYear();
  renderCart();
</script>
